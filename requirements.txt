# GreenTruth runtime dependencies
#
# The engine, the API and the interface use the Python standard library only:
#
#     python server.py        # runs with nothing installed
#
# Without the packages below the app still runs, uses the rule-based claim
# detector and reports that it is the fallback (/api/health, the top bar and
# every result).
#
# The packages below enable the ClimateBERT claim detector
# (https://huggingface.co/Rahilgh/greentruth-claim-detector, public, no token).
# Vercel installs them for the deployed API; locally (Python 3.10+):
#
#     pip install -r requirements.txt
#
# CPU-only PyTorch. The torch wheel comes from PyTorch's CPU download page,
# never the default Linux build, which pulls in several GB of CUDA libraries.
# Every other package comes from PyPI. (macOS wheels are CPU builds without the
# "+cpu" tag.)
--find-links https://download.pytorch.org/whl/cpu/torch/
torch==2.14.0+cpu ; sys_platform != "darwin"
torch==2.14.0 ; sys_platform == "darwin"
transformers==5.17.0
